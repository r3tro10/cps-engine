#pragma once

#include "sensor.hpp"
#include <vector>
#include <memory>

/**
 * @brief Orchestrates the lifecycle and execution of the sensor fleet.
 *
 * ARCHITECTURE: Maintains exclusive ownership of all sensor resources.
 * This ensures a single point of truth for hardware state management.
 */
class SensorManager
{
public:
    SensorManager() = default;
    ~SensorManager() = default;

    /**
     * @brief Ingests a new sensor node into the fleet.
     * @param sensor Unique pointer transferring full ownership to this manager.
     */
    void add_sensor(std::unique_ptr<Sensor> sensor);

    // Triggers a synchronous data acquisition cycle across all managed nodes
    void read_all_sensors();

private:
    // Container uses unique_ptr to enforce strict ownership (no dangling references)
    std::vector<std::unique_ptr<Sensor>> sensors_;
};

#pragma once
#include <string>

class Sensor
{
public:
    // Constructor and Destructor
    Sensor(std::string name);
    ~Sensor(); // Crucial for cleaning up our heap buffer (RAII)

    // --- README Requirement: Non-copyable sensor objects ---
    Sensor(const Sensor &) = delete;            // Disallow copy constructor
    Sensor &operator=(const Sensor &) = delete; // Disallow copy assignment

    // --- Allow Move Semantics (so std::unique_ptr and std::move work) ---
    Sensor(Sensor &&) noexcept = default;
    Sensor &operator=(Sensor &&) noexcept = default;

    void read_telemetry() const;

private:
    std::string name_;

    // --- README Requirement: Allocates a small heap buffer ---
    int *heap_buffer_;
};


===== ./include/isensor.hpp =====

#pragma once

#include <string>

/**
 * @brief Abstract hardware interface layer for physical system telemetries.
 * Forces derived nodes to implement safe lifecycle controls and explicit read procedures.
 */
class ISensor
{
public:
    // 1. Virtual Destructor: Guarantees complete cleanup of derived allocations through base pointers
    virtual ~ISensor() = default;

    // 2. Pure Virtual Interface: Must be implemented by concrete drivers
    // Note: read_telemetry should be non-const because reading hardware often clears registers or updates states
    virtual double read_telemetry() = 0;

    // Note: get_name is const because reading a static string identifier never updates object state
    virtual std::string get_name() const = 0;

    // 3. Non-copyable base to safeguard hardware ownership
    ISensor(const ISensor &) = delete;
    ISensor &operator=(const ISensor &) = delete;

protected:
    // Allow derived classes to be default-constructed or moved safely
    ISensor() = default;
    ISensor(ISensor &&) noexcept = default;
    ISensor &operator=(ISensor &&) noexcept = default;
};

===== ./include/actuators/iactuator.hpp =====

#pragma once

#include "../core/types.hpp"

class IActuator
{
public:
    // Provides the definition of JointState.
    virtual ~IActuator() = default;

    // Without a virtual destructor, the derived destructor may not run correctly.
    virtual void setCommand(double target_effort) = 0;

    // Requires every concrete actuator to implement command handling.
    virtual JointState getState() const = 0;

    // Requires every concrete actuator to provide telemetry, while promising not to modify the actuator during the read.
    virtual void update(double dt) = 0;

    // Requires every concrete actuator to implement time-step processing.
    IActuator(const IActuator &) = delete;
    IActuator &operator=(const IActuator &) = delete;

    // Prevents copying and assigning actuator objects.
protected:
    IActuator() = default;
};

#pragma once

class PIDController
{
    // trailing _ to differentiate private members and parameters
private:
    double kp_;
    double ki_;
    double kd_;

    // min_out must be less than or equal to max_out
    double min_out_;
    double max_out_;

    double integral_err_ = 0.0;
    double prev_measured_ = 0.0;
    bool is_first_step_ = true;

public:
    // Used a member-initiliser then define it in pid.cpp to properly initilise members to prevent undefined behavior
    PIDController(
        double kp,
        double ki,
        double kd,
        double min_out,
        double max_out);

    // dt must be greater than zero and is measured in seconds.
    double compute(double setpoint, double measured, double dt);
};

#include "sensor.hpp"
#include <iostream>

// Constructor: Allocates the small heap buffer
Sensor::Sensor(std::string name) : name_(std::move(name))
{
    // Fulfilling the README: Allocating a small heap buffer (e.g., space for 10 integers)
    heap_buffer_ = new int[10]{0};

    std::cout << "[Hardware Connected] " << name_
              << " initialized. Heap buffer allocated at: " << heap_buffer_ << "\n";
}

// Destructor: Cleans up the buffer automatically when the object falls out of scope
Sensor::~Sensor()
{
    if (heap_buffer_ != nullptr)
    {
        delete[] heap_buffer_; // RAII: Fulfilling the "Automatic cleanup" promise
        std::cout << "[Hardware Disconnected] " << name_ << " released dynamic heap buffer.\n";
    }
}

void Sensor::read_telemetry() const
{
    std::cout << "[Telemetry] Reading from " << name_ << "...\n";
}

===== ./src/sensor_manager.cpp =====

#include "sensor_manager.hpp"

void SensorManager::add_sensor(std::unique_ptr<Sensor> sensor)
{
    if (sensor)
    {
        // Move semantics: Transfer ownership from caller (Stack) to Manager (Vector)
        sensors_.push_back(std::move(sensor));
    }
}

void SensorManager::read_all_sensors()
{
    // Iterate through the fleet to execute the control loop cycle
    for (const auto &sensor : sensors_)
    {
        sensor->read_telemetry();
    }
}


===== ./src/main.cpp =====

#include "sensor.hpp"
#include "sensor_manager.hpp"
#include "actuators/mock_actuator.hpp"
#include "control/pid.hpp"

#include <chrono>
#include <iostream>
#include <memory>
#include <thread>

int main()
{
    std::cout << "=== CPS Engine: Boot Sequence ===\n\n";

    auto lidar = std::make_unique<Sensor>("LiDAR_Front");
    auto imu = std::make_unique<Sensor>("IMU_Primary");
    auto camera = std::make_unique<Sensor>("Depth_Cam_0");

    SensorManager manager;

    std::cout << "\n[LIFECYCLE] Transferring ownership to SensorManager...\n";

    manager.add_sensor(std::move(lidar));
    manager.add_sensor(std::move(imu));
    manager.add_sensor(std::move(camera));

    std::cout << "\n[EXECUTION] Starting telemetry acquisition cycle:\n";
    manager.read_all_sensors();

    PIDController pid(15.0, 0.0, 2.0, -10.0, 10.0);
    MockActuator motor;

    constexpr double target_rad = 1.57;
    constexpr double dt = 0.02;

    using clock = std::chrono::steady_clock;
    const auto loop_period = std::chrono::milliseconds(20);
    auto next_tick = clock::now();

    for (int step = 0; step < 300; ++step)
    {
        next_tick += loop_period;

        motor.update(dt);
        JointState state = motor.getState();

        double effort = pid.compute(target_rad, state.position, dt);
        motor.setCommand(effort);

        std::cout << "[Step " << step << "] "
                  << "Pos: " << state.position << " rad | "
                  << "Err: " << target_rad - state.position << " rad | "
                  << "Effort: " << effort << "\n";

        std::this_thread::sleep_until(next_tick);
    }

    std::cout << "\n[SHUTDOWN] System halt. Initiating RAII teardown...\n";

    return 0;
}

===== ./src/control/pid.cpp =====

#include "control/pid.hpp"
#include <algorithm>

double PIDController::compute(double setpoint, double measured, double dt)
{
    if (dt <= 0.0)
    {
        return 0.0; // Fail-safe: zero effort if time hasn't advanced
    }

    const double error = setpoint - measured;

    integral_err_ += error * dt;
    integral_err_ = std::clamp(integral_err_, min_out_, max_out_); // Prevent integral windup for when errors accumulate very quickly

    double derivative = 0.0;
    if (is_first_step_)
    {
        is_first_step_ = false;
    }
    else
    {
        derivative = -(measured - prev_measured_) / dt;
    }

    const double raw_output = (kp_ * error) + (ki_ * integral_err_) + (kd_ * derivative);

    prev_measured_ = measured;

    return std::clamp(raw_output, min_out_, max_out_);
}

PIDController::PIDController(
    double kp,
    double ki,
    double kd,
    double min_out,
    double max_out)
    : kp_(kp),
      ki_(ki),
      kd_(kd),
      min_out_(min_out),
      max_out_(max_out)
{
}
